2. ESTRUTURA IF

numero = 10

if numero > 5:
    resultado = numero


3. ESTRUTURA IF E ELSE

numero = 3

if numero > 5:
    resultado = 1
else:
    resultado = 0


4. ESTRUTURA WHILE

contador = 5

while contador > 0:
    contador = contador - 1


5. ESTRUTURA FOR

    resultado = i

8. DEFINICAO DE FUNCAO COM DOIS PARAMETROS

9. OPERADORES RELACIONAIS

a = 10
b = 5

if a == b:
    resultado = 1

if a != b:
    resultado = 2

if a < b:
    resultado = 3

if a > b:
    resultado = 4

if a <= b:
    resultado = 5

if a >= b:
    resultado = 6
